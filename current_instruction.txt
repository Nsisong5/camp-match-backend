from __future__ import annotations
from collections.abc import AsyncGenerator
from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession, async_sessionmaker, create_async_engine
from camp_match.config.settings import Settings

def create_engine(settings: Settings) -> AsyncEngine:
    return create_async_engine(
        settings.database_url,
        pool_size=settings.database_pool_size,
        echo=settings.database_echo,
    )

def create_session_factory(engine: AsyncEngine) -> async_sessionmaker[AsyncSession]:
    return async_sessionmaker(bind=engine, expire_on_commit=False)

Switch create_app() to use FastAPI's lifespan parameter:

from contextlib import asynccontextmanager
from fastapi import FastAPI
from camp_match.config.settings import get_settings
from camp_match.platform.db.session import create_engine, create_session_factory

@asynccontextmanager
async def lifespan(app: FastAPI):
    settings = get_settings()
    engine = create_engine(settings)
    app.state.db_engine = engine
    app.state.db_session_factory = create_session_factory(engine)
    yield
    await engine.dispose()

def create_app() -> FastAPI:
    app = FastAPI(title="Camp Match API", version="0.1.0", lifespan=lifespan)

CHUNK 3.4 — Repository Foundation (Unit of Work
Objective: A concrete SqlAlchemyUnitOfWork satisfying the UnitOfWork port defined in Phase 2 Chunk 2.5 — exactly, not a redesign of it.
Files to create:
src/camp_match/platform/db/unit_of_work.py
Implementation instructions:

from __future__ import annotations
from sqlalchemy.ext.asyncio import AsyncSession

class SqlAlchemyUnitOfWork:
    def __init__(self, session: AsyncSession) -> None:
        self._session = session

    async def __aenter__(self) -> "SqlAlchemyUnitOfWork":
        return self

    async def __aexit__(self, exc_type: object, exc: object, tb: object) -> None:
        if exc_type is not None:
            await self.rollback()

    async def commit(self) -> None:
        await self._session.commit()

    async def rollback(self) -> None:
        await self._session.rollback()

    @property
    def session(self) -> AsyncSession:
        """Exposed so a concrete repository (built alongside its owning module) can execute queries within this unit of work's transaction. Only that module's own repository implementation should use this — never a use case directly."""
        return self._session
        
Add a FastAPI dependency get_unit_of_work in the same file that composes get_db_session → SqlAlchemyUnitOfWork, for use by future module composition roots.
Repository pattern conventions (document only — do not implement a generic repository class): each module defines its own repository interface in its application/ports/outbound.py (e.g. ListingRepository with methods meaningful to Housing specifically — find_by_id, find_published_near, etc. — not a generic save/get/delete CRUD interface shared across modules), and its own implementation in adapters/persistence/repository.py, constructed with a Session or the UnitOfWork. Do not create a shared GenericRepository[T] base class — that pattern hides module-specific query needs behind a false abstraction and is exactly the kind of shared-kernel sprawl Phase 2 was designed to avoid. Add this paragraph to AI_instructions/07_DATABASE_RULES.md.
